def to_decimal(value: Any, default: Decimal = Decimal('0')) -> Decimal:
if value is None or value == "":
return default
try:
s = str(value).replace(',', '.')
return Decimal(s)
except (InvalidOperation, ValueError):
return default

if value is None or value == "":
return default
try:
return float(Decimal(str(value).replace(",", ".").strip()))
except (InvalidOperation, ValueError):
return default

except (ValueError, TypeError):
return default

def extract_tariff_data_from_rows(rows: List[Dict[str, Any]]) -> Dict[str, Any]:
"""
Агрегирует данные. Теперь ищет сумму даже если в колонке
'with_discount' стоит 0,
проверяя другие колонки в сырой строке (если бы мы парсили сырую
строку),
но так как мы парсим уже разбитые строки, мы будем искать по
названию.
"""
plan_fee = Decimal('0')
daily_fee = Decimal('0')
days_count = 0
found_plan = False
found_daily = False
for row in rows:
# Проверяем сумму в 'with_discount'
cost_str = row.get('with_discount', '0')
cost = to_decimal(cost_str)
# Если сумма 0, пробуем найти её в 'no_discount' или 'discount' (на
случай ошибки парсера)
if cost == 0:
no_disc = to_decimal(row.get('no_discount', '0'))
disc = to_decimal(row.get('discount', '0'))
if no_disc > 0:
cost = no_disc
elif disc > 0:
cost = disc
volume = row.get('volume', '')
service_name = row.get('service', '').lower()
is_plan = False
is_daily = False
if 'абонентская плата по тарифному плану' in service_name:
if 'посуточное' in service_name:
is_daily = True
found_daily = True
else:is_plan = True
found_plan = True
if is_daily:
try:
vol_str = str(volume).lower().replace(' шт', '').replace('шт', '').strip()
if vol_str:
days_count = int(float(vol_str))
except ValueError:
pass
daily_fee += cost
elif is_plan:
plan_fee += cost
if not found_plan and not found_daily:
print(f"[DEBUG EXTRACT] Ничего не найдено по названию! Услуги:
{[r.get('service', 'N/A') for r in rows[:3]]}")
elif found_plan or found_daily:
print(f"[DEBUG EXTRACT] Найдено: Plan={plan_fee}, Daily={daily_fee},
Days={days_count}")
return {
'plan_fee': plan_fee,
'daily_fee': daily_fee,
'days': days_count if days_count > 0 else 30
}
