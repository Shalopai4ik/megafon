def convert_to_utf8(input_file, output_file=None):
    """
    Преобразует файл в UTF-8, автоматически определяя исходную кодировку
    """
    if output_file is None:
        output_file = input_file.replace('.csv', '_utf8.csv')
    # Сначала пробуем определить кодировку с помощью file
    try:
        result = subprocess.run(['file', '-i', input_file],
                              capture_output=True, text=True)
        mime_info = result.stdout.lower()
        # Если уже UTF-8, просто копируем
        if 'utf-8' in mime_info:
            if input_file != output_file:
                subprocess.run(['cp', input_file, output_file])
            return output_file
    except:
        pass
    # Пробуем различные кодировки
    encodings = ['cp1251', 'cp866', 'koi8-r', 'iso-8859-5']
    for encoding in encodings:
        try:
            result = subprocess.run([
                'iconv', '-f', encoding, '-t', 'utf-8', input_file
            ], capture_output=True, text=True)
            if result.returncode == 0:
                with open(output_file, 'w', encoding='utf-8') as f:
                    f.write(result.stdout)
                print(f"✅ Файл {input_file} успешно преобразован из {encoding} в UTF-8")
                return output_file
        except:
            continue
    # Если ничего не помогло, используем универсальный подход
    try:
        # Читаем с игнорированием ошибок и сохраняем как UTF-8
        with open(input_file, 'r', encoding='utf-8', errors='ignore') as f:
            content = f.read()
        with open(output_file, 'w', encoding='utf-8') as f:
            f.write(content)
        print(f"⚠️ Файл {input_file} преобразован с игнорированием ошибок")
        return output_file
    except Exception as e:
        print(f"❌ Не удалось преобразовать {input_file}: {e}")
        return input_file  # Возвращаем оригинальный файл

def clean_line(line):
    """
    Очищает строку от ненужных элементов и нормализует пробелы
    """
    # Удаляем паттерн "[число] из [число]"
    line = re.sub(r'\d+\s+из\s+\d+', '', line, flags=re.IGNORECASE)

    # Заменяем все запятые на точки
    line = line.replace(',', '.')

    # Заменяем множественные пробелы на одинарные
    line = re.sub(r'\s+', ' ', line)

    # Убираем лишние пробелы в начале и конце
    line = line.strip()

    return line

// ошибку с понятным текстом (её ловят вызывающие и показывают баннер).

async function getJSON(url) {
const resp = await fetch(url, { headers: { Accept: 'application/json' } });
let data = null;
try { data = await resp.json(); } catch (_) { /* тело не JSON */ }
if (!resp.ok) {
const msg = (data && data.error) ? data.error :  HTTP ${resp.status} ;
throw new Error(msg);
}
return data;
}

function toggleTheme() {
const root = document.documentElement;
const isDark = root.getAttribute('data-theme') === 'dark'
|| (!root.getAttribute('data-theme') && matchMedia('(prefers-color-scheme: dark)').matches);
const next = isDark ? 'light' : 'dark';
root.setAttribute('data-theme', next);
localStorage.setItem('theme', next);
if (document.getElementById('chartPanel').classList.contains('open') && trendCache) drawBigChart();
}

function renderUsers() {
const grid = document.getElementById('usersGrid');
const term = document.getElementById('searchInput').value.trim().toLowerCase();
let list = subscribers.filter((s) => {
if (term && !(String(s.number).includes(term)  (s.username  '').toLowerCase().includes(term))) return false;
if (currentFilter === 'all') return true;
if (currentFilter === 'overpay') return s.overpayment > 0;
if (currentFilter === 'underused') return s.underused;
return s.status === currentFilter;
});
const dir = sortDirection === 'desc' ? -1 : 1;
const keys = {
overpay: (a, b) => a.overpayment - b.overpayment,
total: (a, b) => a.total - b.total,
risk: (a, b) => a.risk - b.risk,
number: (a, b) => String(a.number).localeCompare(String(b.number)),
};
list.sort((a, b) => dir * (keys[currentSort] || keys.overpay)(a, b));
document.getElementById('resultCount').innerText =  ${list.length} из ${subscribers.length} ;
if (!list.length) { grid.innerHTML = '<div class="empty-state">Нет данных для отображения</div>'; return; }
grid.innerHTML = list.map(renderCard).join('');
}
